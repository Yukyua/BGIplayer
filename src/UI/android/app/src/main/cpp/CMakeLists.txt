cmake_minimum_required(VERSION 3.22)
project(bgiplayer_android LANGUAGES C CXX)
get_filename_component(BGI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../../.." ABSOLUTE)
set(NATIVE "${BGI_ROOT}/build/android-native")
set(PREFIX "${NATIVE}/arm64-v8a/prefix")
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${NATIVE}/sources/SDL2-2.32.10" "${CMAKE_BINARY_DIR}/sdl")
file(GLOB ENGINE_SOURCES CONFIGURE_DEPENDS "${BGI_ROOT}/src/*.cpp")
add_library(bgiplayer SHARED ${ENGINE_SOURCES} android_host.cpp)
set_source_files_properties("${BGI_ROOT}/src/main.cpp" PROPERTIES COMPILE_OPTIONS "-include;SDL_main.h")
target_compile_features(bgiplayer PRIVATE cxx_std_20)
target_compile_options(bgiplayer PRIVATE -O2 -fvisibility=hidden)
target_include_directories(bgiplayer PRIVATE "${BGI_ROOT}/include" "${PREFIX}/include" "${PREFIX}/include/freetype2")
target_link_options(bgiplayer PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(bgiplayer PRIVATE SDL2 SDL2main
 "${PREFIX}/lib/libavformat.a" "${PREFIX}/lib/libavcodec.a" "${PREFIX}/lib/libswscale.a" "${PREFIX}/lib/libswresample.a" "${PREFIX}/lib/libavutil.a" "${PREFIX}/lib/libfreetype.a" "${PREFIX}/lib/libiconv.a" EGL GLESv3 android log z m)
